        function updateDebtField(bank, field, value) {

            if (!appData.debts[bank]) {

                appData.debts[bank] = { current: 0, offer: 0, location: "" };

            }

            if (field === 'current' || field === 'offer') {

                appData.debts[bank][field] = value === "" ? 0 : parseFloat(value);

            } else {

                appData.debts[bank][field] = value;

            }

            saveStorage();


            // Atualiza apenas o total de dívidas sem re-renderizar todo o input e perder foco

            let totalDebts = 0;

            Object.keys(appData.debts).forEach(b => {

                totalDebts += Number(appData.debts[b].current) || 0;

            });

            document.getElementById('total-debts-label').innerText = formatMoney(totalDebts);

        }

        function renderMonthsList() {

            container.innerHTML = keys.map(key => {

                const exp = monthData.expenses.reduce((s, e) => s + Number(e.amount), 0);

                const isActive = key === appData.activeMonth;


                return `

                            </div>

                            <div>

                                <h4 class="text-xs font-bold text-stone-800">${getMonthLabel(key)}</h4>

                                <span class="text-[10px] text-stone-400">Gastos: ${formatMoney(exp)}</span>

                            </div>

                        </div>

                        <div class="flex items-center space-x-1.5">

                            <button onclick="switchMonth('${key}')" class="px-2.5 py-1 rounded-lg ${isActive ? 'bg-pink-500 text-white' : 'border border-rose-border text-stone-600 hover:bg-pink-50'} text-[11px] font-semibold transition-all">

                                ${isActive ? 'Ativo' : 'Selecionar'}

                            </button>

                            ${keys.length > 1 ? `

                                <button onclick="deleteMonth('${key}')" class="p-1.5 text-stone-300 hover:text-rose-500 transition-colors" title="Excluir Mês">

                                    <i class="fa-solid fa-trash-can text-[10px]"></i>

                `;

            }).join('');

        }

        function createNewMonth() {

            const val = document.getElementById('new-month-input').value; // YYYY-MM

            if (!val) return;


            if (!appData.monthsData[val]) {

                appData.monthsData[val] = {

                    incomes: [

                        { id: 1, desc: "Salário Fixo", amount: 3650.00 },

                        { id: 2, desc: "VR (Vale-Refeição)", amount: 550.00 }

                    ],

                    expenses: [],

                    investments: []

                };

            }

            appData.activeMonth = val;

            saveStorage();

            updateUI();

            renderMonthsList();

        }


        function deleteMonth(key) {

            if (Object.keys(appData.monthsData).length <= 1) return;

            delete appData.monthsData[key];

            if (appData.activeMonth === key) {

                appData.activeMonth = Object.keys(appData.monthsData)[0];

            }

            saveStorage();

            updateUI();

            renderMonthsList();

        }
